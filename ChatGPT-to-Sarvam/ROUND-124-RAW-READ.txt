2: ; ROUND-45: --debug, the smallest useful debugger.
3: ;
4: ;   sutram_compiler --debug <in.sm> <out>
5: ;   sutram_compiler --debug --lang <pak> <in.sm> <out>
6: ;   sutram_compiler --debug=<name>[,<name>...][@N[-M]] <in.sm> <out>      (R46)
7: ;
8: ; R46: --debug=yoga,i shows only those names, and prints no line where none of
9: ; them is in scope; @N or @N-M prints only statements on lines N.. or N..M
10: ; (every statement is still counted). At the end of mukhya's body the program
11: ; prints "[shodhan] samapti, vakya: <statements run>" and every scalar then in
12: ; scope. The counter is a module-level vitti registered before parsing, named
13: ; so that a program cannot write it (0shodhan_vakya), incremented by an
14: ; ordinary generated `c = c + 1` before each statement. Module-level vitti are
15: ; shown in the trace too (after the locals), unless a var_table entry of the
16: ; same name would be what the name reads. The summary is not printed when the
17: ; program ends early (prati in mukhya, sthagita, a runtime error).

465: ; R46: rdi = an argument. rax = 0 if it is not --debug; 1 if it is --debug or
466: ; --debug=<spec> (r45_debug set, the spec read); 2 if the spec is malformed.
467: ;   <spec> = [name{,name}][@N[-M]]   e.g. yoga,i   yoga@20-40   @20
468: ; The names and numbers are cut out of the argument in place.
469: r45_match_flag:
470:     push rbx
471:     mov rbx, rdi
472:     lea rsi, [rel r45_flag]        ; "--debug"
473: .pre:
474:     mov al, [rsi]
475:     test al, al
476:     jz .pre_done
477:     cmp al, [rbx]
478:     jne .no
479:     inc rsi
480:     inc rbx
481:     jmp .pre
482: .pre_done:
483:     mov al, [rbx]
484:     test al, al
485:     jz .yes                        ; exactly --debug
486:     cmp al, '='
487:     jne .no                        ; --debugx: not the flag
488:     inc rbx
489:     cmp byte [rbx], 0
490:     je .bad                        ; --debug= with nothing
491: .name:                             ; a name runs to ',', '@' or the end
492:     mov al, [rbx]
493:     cmp al, '@'
494:     je .range
495:     test al, al
496:     jz .yes
497:     cmp al, ','
498:     je .bad                        ; an empty name
499:     mov rcx, [rel r45_watch_cnt]
500:     cmp rcx, R45_WATCH_CAP
501:     jae .bad
502:     lea rdx, [rel r45_watch]
503:     mov [rdx + rcx*8], rbx
504:     inc qword [rel r45_watch_cnt]
505: .name_end:
506:     mov al, [rbx]
507:     test al, al
508:     jz .yes
509:     cmp al, '@'
510:     je .cut_range
511:     cmp al, ','
512:     je .cut_next
513:     inc rbx
514:     jmp .name_end
515: .cut_next:
516:     mov byte [rbx], 0
517:     inc rbx
518:     cmp byte [rbx], 0
519:     je .bad                        ; a trailing ','
520:     jmp .name
521: .cut_range:
522:     mov byte [rbx], 0
523: .range:
524:     inc rbx                        ; past '@'
525:     call .number
526:     test rax, rax
527:     jz .bad
528:     mov [rel r45_from], rax
529:     cmp byte [rbx], 0
530:     je .yes
531:     cmp byte [rbx], '-'
532:     jne .bad
533:     inc rbx
534:     call .number
535:     test rax, rax
536:     jz .bad
537:     cmp rax, [rel r45_from]
538:     jb .bad
539:     mov [rel r45_to], rax
540:     cmp byte [rbx], 0
541:     jne .bad
542: .yes:
543:     mov qword [rel r45_debug], 1
544:     mov eax, 1
545:     pop rbx
546:     ret
547: .no:
548:     xor eax, eax
549:     pop rbx
550:     ret
551: .bad:
552:     mov eax, 2
553:     pop rbx
554:     ret

=== docs/LANGUAGE-REFERENCE.md blob=86bc60620bde169a2136cd6b1e934323f94d59bb original_lines=471-495 ===
471: `a[lo:hi]` is a view of the elements `lo .. hi-1` of a `mala`, an integer `kosh`
472: or another slice. **Nothing is copied:** a write through the slice changes the
473: array. `a[lo:]` runs to the end, `a[:hi]` starts at 0, `a[:]` is all of it.
474: 
475: ```
476: prakriya yog(vibhag s) { ... pankti_len(s) ... s[i] ... }
477: likha(yog(a[2:5]))         # elements 2, 3, 4
478: vitti s = a[1:7]           # s is a slice: s[0] is a[1]
479: likha(yog(a))              # a whole mala or kosh is passed as a[:]
480: ```
481: 
482: - **Checked twice at run time:** when made (`0 <= lo <= hi <= length`) and when
483:   indexed (`s[i]`, `s[i] = v` against the slice's own length), with the array
484:   bounds diagnostic, exit 1.
485: - **`vibhag` (Devanagari `विभाग`, and a word in each pack) declares a slice
486:   parameter.** It takes a slice, or a whole `mala` or integer `kosh`; anything
487:   else is a compile error (`vibhag: mala, kosh va vibhag apekshita`), as is
488:   slicing a raw pointer (an untyped parameter, a `nirmmita` result).
489: - **A variable is a slice when it is declared from one** (`vitti s = a[1:3]`,
490:   `vitti t = s`) or is a `vibhag` parameter.
491: - **A slice lives in the frame of the function that makes it:** returning one is a
492:   compile error (`vibhag prakriyayah na pratyarpaniyah`). A slice made at one place
493:   in a loop is the same slice each time round (the next iteration remakes it).
494: - `pankti_len(s)` is the slice's length.
