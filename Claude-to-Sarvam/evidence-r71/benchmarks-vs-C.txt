## bench_fib.c
#include <stdio.h>
long fib(long n){if(n<2)return n;return fib(n-1)+fib(n-2);}

## bench_nested.c

int main(){long t=0,i=0;while(i<2000){long j=0;while(j<2000){if((i+j)%3==0)t=t+1;j=j+1;}i=i+1;}printf("%ld\n",t);}

## r46_hotcall_array.c
#include <stdio.h>
long leaf(long x){return x+3;} long bridge(long x){return leaf(x);}
int main(){long nums[4]={7,11,13,17},i=0,sum=0;while(i<3000000){sum=sum+bridge(nums[i%4]);i=i+1;}printf("%ld\n",sum);}

## r46_hotcall_float.c

## r46_hotcall_nested.c
#include <stdio.h>

int main(){long i=0,sum=0;while(i<4500000){sum=sum+outer(i);i=i+1;}printf("%ld\n",sum);}
