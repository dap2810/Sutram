// R76 estimate: cost of a shadow-table bounds check per indexed write, the raw_write pattern
// (1024 cells x 20000 passes), with the table holding N live allocations (bump order = sorted).
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <time.h>
typedef struct { uintptr_t lo, hi; } R;
static R *tab; static long ntab;
static inline void check(uintptr_t p) {           // find the allocation containing p
    long lo = 0, hi = ntab - 1;
    while (lo <= hi) { long m = (lo + hi) >> 1;
        if (p < tab[m].lo) hi = m - 1; else if (p + 8 > tab[m].hi) lo = m + 1; else return; }
    fprintf(stderr, "oob\n"); exit(1);
}
static double now(void){ struct timespec t; clock_gettime(CLOCK_MONOTONIC,&t); return t.tv_sec+t.tv_nsec*1e-9; }
int main(int argc, char **argv) {
    long N = atol(argv[1]); int checked = atoi(argv[2]);
    char *arena = malloc((N + 1) * 64 + 8192 * 2); uintptr_t cur = ((uintptr_t)arena + 15) & ~15;
    tab = malloc(sizeof(R) * (N + 1));
    for (long i = 0; i < N; i++) { tab[i].lo = cur; tab[i].hi = cur + 48; cur += 64; }
    long *b = (long *)cur; tab[N].lo = cur; tab[N].hi = cur + 1024 * 8; ntab = N + 1;
    volatile long sink = 0; double t = now();
    for (long r = 0; r < 20000; r++) for (long j = 0; j < 1024; j++) {
        if (checked) check((uintptr_t)&b[j]);
        b[j] = j + r; }
    sink = b[1023]; double e = now() - t;
    printf("N=%-8ld %s %.4f s  %.2f ns/write\n", N, checked ? "checked  " : "unchecked", e, e * 1e9 / (20000.0 * 1024));
    return (int)(sink & 0);
}
