--- libsweep.py
# R80 WS3: every library function whose parameters are all plain numbers, called with plausible
# non-negative inputs; flag any call that returns a NEGATIVE number (a wrapped result).
import sys, itertools, tempfile
from pathlib import Path
sys.path.insert(0, 'tools')
import lib_mutant_witness as W, lib_proof_gaps as g
comp = str(Path(sys.argv[1]).resolve())
funcs = {(f['module'], f['name']): f for p in sorted(g.LIB.glob('*.smlib')) for f in g.parse_module(p)}
ptrs = W.pointer_params(funcs)
V1 = [0, 1, 2, 3, 5, 10, 12, 15, 20, 21, 25, 30, 40, 50, 63, 64, 66, 67, 70, 93, 100, 1000, 10**4, 10**5, 10**6, 10**9, 3037000500, 10**10, 10**12]
V2 = [0, 1, 2, 3, 7, 10, 12, 20, 63, 64, 100, 1000, 10**6, 10**9, 10**10]
base = Path(tempfile.mkdtemp()); g.make_tree(base)
flagged = []; ntried = 0
for (mod, name), f in sorted(funcs.items()):
    if ptrs[(mod, name)] or 'dasham' in f['params']: continue
    ar = len([p for p in f['params'].split(',') if p.strip()])
    if ar == 0 or ar > 3: continue
    header, _ = g.probe_header(comp, base, mod)
    if header is None: continue
    vals = V1 if ar == 1 else V2 if ar == 2 else [0, 1, 2, 10, 100, 10**6, 10**9]
    args = list(itertools.product(vals, repeat=ar))
    out = W.run(comp, base, header, name, args)
    ntried += 1
    neg = [(a, o) for a, o in zip(args, out or []) if o is not None and o.lstrip('-').isdigit() and int(o) < 0]
    if neg:
        flagged.append((mod, name, f['params'], len(neg), len(args), neg[:4]))
print(f'{ntried} number-only library functions swept; {len(flagged)} return a negative number for some non-negative input:')
for m, n, p, k, tot, ex in flagged:
    print(f'  {m}.{n}({p}): {k}/{tot} calls negative; e.g. ' + '; '.join(f'{n}{a} = {o}' for a, o in ex))
