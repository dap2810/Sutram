== a kosh made in a block given to a prakriya that only reads it
prakriya yoga(kosh k) {
    vitti s = 0
    chakra i = 0 tak kosh_gan(k) - 1 {
        s = s + k[i]
    }
    nivartanam s
}
mukhya() {
    mandapa {
        kosh k[2]
        kosh_kship(k, 3)
        kosh_kship(k, 4)
        likha(yoga(k))
    }
}
-- compile: Siddh: 3545 baait sankalit
-- output: 7

== ... and to one that keeps it (returns it)
prakriya rakh(kosh k) {
    nivartanam k
}
mukhya() {
    mandapa {
        kosh k[2]
        kosh_kship(k, 3)
        vitti j = rakh(k)
        likha(kosh_gan(j))
    }
}
-- compile: Sutram truti [E_KOSH_MANDAPA]: mandape rachitah kosh tam prakriyam na gacchati ya tam rakshati (nivartayati, sthapayati, anyatra preshayati)
