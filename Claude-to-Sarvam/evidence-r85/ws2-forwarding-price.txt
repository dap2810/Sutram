R85 WS2 - the price of a forwarding pointer, measured WITHOUT building it into the compiler: the k[i] read sequence v278 emits,
copied into a standalone NASM loop, with and without the 3-instruction forward check (mov rdx,[k-24] ; test rdx,rdx ; jnz cold);
'call' wraps the read in a prakriya with R82's entry check, as padh(kosh k, i) in R82's worst case.

The host is shared and noisy (see the spreads). By medians +7.7% / +14.6%; by best runs +16% / +8%. Read it as: 3 instructions,
0.1-0.4 ns per access, 8-16% on a loop that does nothing but read a kosh.

--- read2.mac (the read, with FWD) and call.asm (the harness)
  | %macro call0 1
  | %if FWD
  |     mov rdx, [rax-24]              ; forward word
  |     test rdx, rdx
  |     jnz follow_cold                ; out of line: never taken here
  | %endif
  |     push rax
  |     mov rcx, [rax-16]
  |     push rcx
  |     mov rax, rsi
  |     cmp rax, [rsp]
  |     jb %%ok
  |     ud2
  | %%ok:
  |     pop rcx
  |     shl rax, 3
  |     pop rcx
  |     add rax, rcx
  |     mov rax, [rax]
  | %ifidn %1, rax
  | %else
  |     add %1, rax
  | %endif
  | 
  | %endmacro
  | ; R85 WS2 price: the kosh read sequence v278 emits, in a loop - fwd=0 adds the 3-instruction forward check;
  | ; call=1 wraps it in a call with R82's entry check (as padh(kosh k, i)).
  | global _start
  | section .bss
  |     buf resq 64
  | section .text
  | _start:
  |     lea rbx, [rel buf+40]          ; k: user pointer, header below it
  |     mov qword [rbx-16], 16         ; len
  |     mov qword [rbx-24], 0          ; forward word: 0 = not moved (fwd variant reads it)
  |     mov qword [rbx-32], 0          ; R82 slot: 0 = made outside a block
  |     mov r14, 200000000
  |     xor r12, r12
  |     xor r13, r13
  | .loop:
  | %if 1
  |     mov rdi, rbx
  |     mov rsi, r12
  |     call padh
  |     add r13, rax
  | %else
  |     mov rax, rbx
  |     mov rsi, r12
  |     call0 r13
  | %endif
  |     inc r12
  |     and r12, 15
  |     dec r14
  |     jnz .loop
  |     mov eax, 60
  |     xor edi, edi
  |     syscall
  | %if 1
  | padh:
  |     push rbp
  |     mov rbp, rsp
  |     push rbx
  |     push rdi
  |     push rsi
  |     mov rax, rdi                   ; R82 entry check (flag-on), as emitted
  |     test rax, rax
  |     jz .ok
  |     mov rcx, [rax-32]
  |     test rcx, rcx
  |     jz .ok
  |     cmp rcx, rsp
  |     jb .ok
  |     mov rdx, [rcx]
  |     cmp rdx, [rax-24]
  |     je .ok
  | .ok:
  |     mov rax, [rbp-16]
  |     mov rsi, [rbp-24]
  |     call0 rax
  |     lea rsp, [rbp-8]
  |     pop rbx
  |     pop rbp
  |     ret
  | %endif
